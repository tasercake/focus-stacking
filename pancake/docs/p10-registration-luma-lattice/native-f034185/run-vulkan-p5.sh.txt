#!/usr/bin/env bash
set -euo pipefail
record=$(realpath "$1")
test -s "$record/vulkan-p5-quiet-lease.txt"
test "$(git rev-parse HEAD)" = f034185843cb20bfb61c31655f999e2f26e1c939
test -z "$(git status --porcelain)"
(cd pancake && sha256sum -c "$record/lattice-release-sha256.txt")
mkdir -m 700 "$record/vulkan-xdg"
unset DISPLAY WAYLAND_DISPLAY DBUS_SESSION_BUS_ADDRESS PANCAKE_RUNTIME
unset VK_ICD_FILENAMES VK_ADD_DRIVER_FILES LD_LIBRARY_PATH
export XDG_RUNTIME_DIR="$record/vulkan-xdg"
export VK_DRIVER_FILES=/usr/share/vulkan/icd.d/nvidia_icd.json
gpu_test=$(cat "$record/lattice-release-gpu-test-path.txt")
adapter=pancake-conformance/src/pancake_conformance/adapters/pancake/pancake.toml
runner="$record/conformance-venv/bin/pancake-conformance"
trap 'result=$?;
  { TZ=Asia/Singapore date --iso-8601=seconds; printf "lane_exit=%s\n" "$result"; } > "$record/vulkan-p5-terminal.txt"
' EXIT
TZ=Asia/Singapore date --iso-8601=seconds > "$record/vulkan-p5-start.txt"
nvidia-smi --query-gpu=name,uuid,pci.bus_id,driver_version --format=csv,noheader > "$record/vulkan-p5-nvidia.txt"
vulkaninfo --summary > "$record/vulkan-p5-vulkaninfo.txt" 2> "$record/vulkan-p5-vulkaninfo-stderr.txt"
grep -F 'NVIDIA GeForce RTX 3090' "$record/vulkan-p5-nvidia.txt"
grep -F 'NVIDIA GeForce RTX 3090' "$record/vulkan-p5-vulkaninfo.txt"
test "$(grep -c '^GPU[0-9]' "$record/vulkan-p5-vulkaninfo.txt")" = 1
native=(timeout --signal=TERM --kill-after=5s 300s taskset -c 2,3
  "$gpu_test" --exact gpu_registration_geometry --test-threads 1)
printf '%q ' "${native[@]}" > "$record/vulkan-p5-native.command"
printf '\n' >> "$record/vulkan-p5-native.command"
"${native[@]}" > "$record/vulkan-p5-native.log" 2>&1

run_case_set() {
  local lane=$1 cases=$2
  local -a options=()
  local case_id
  while IFS= read -r case_id; do options+=(--case "$case_id"); done < "$cases"
  local -a run=(timeout --signal=TERM --kill-after=5s 420s taskset -c 2,3
    "$runner" run --target "$adapter" --no-coverage
    "${options[@]}" --output "$record/p5-$lane")
  printf '%q ' "${run[@]}" > "$record/p5-$lane.command"
  printf '\n' >> "$record/p5-$lane.command"
  "${run[@]}" > "$record/p5-$lane.log" 2>&1
}
printf '%s\n' algorithms.recovery.cascade.gpu > "$record/p5-vulkan-case.txt"
run_case_set vulkan "$record/p5-vulkan-case.txt"
TZ=Asia/Singapore date --iso-8601=seconds > "$record/vulkan-p5-physical-terminal.txt"
run_case_set host "$record/p5-cpu-cases.txt"
python3 "$record/audit-p5.py" "$record"
(cd pancake && sha256sum -c "$record/lattice-release-sha256.txt")
test -z "$(git status --porcelain)"
