"""Compare frozen source photos under the historical and corrected references."""

import hashlib
import json
import subprocess
from pathlib import Path

import numpy as np
import rawpy
from pancake_conformance.adapters.pancake.cr3_reference import decode, geometry

scratch = Path("artifacts/cr3-reference")
manifest = json.loads(
    Path(
        "pancake-conformance/src/pancake_conformance/adapters/pancake/performance-raw-corpus.json"
    ).read_bytes()
)


def sha(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def rgb_sha(pixels):
    return hashlib.sha256(np.ascontiguousarray(pixels, dtype="<u2")).hexdigest()


def measures(candidate, reference):
    delta = np.subtract(candidate, reference, dtype=np.float32)
    bias = delta.mean(axis=(0, 1), dtype=np.float64) / 65535
    absolute = np.abs(delta, out=delta)
    return {
        "mean_absolute_normalized_rgb": float(absolute.mean(dtype=np.float64) / 65535),
        "p99_absolute_normalized_rgb": float(
            np.quantile(absolute, 0.99, overwrite_input=True) / 65535
        ),
        "max_absolute_normalized_rgb": float(absolute.max() / 65535),
        "channel_signed_mean_normalized_rgb": bias.tolist(),
    }


result = {"frames": {}, "orientation": {}}
for index in (0, 49):
    frozen = manifest["frames"][index]
    path = scratch / "sources" / Path(frozen["source_path"]).name
    assert sha(path) == frozen["sha256"] and path.stat().st_size == frozen["size_bytes"]
    output = scratch / "rawler.pixels"
    try:
        subprocess.run(
            [
                str(scratch / "probe/target/debug/pancake-cr3-probe"),
                str(path),
                "default",
                str(output),
                "0,0,0,0",
            ],
            check=True,
            capture_output=True,
        )
        with output.open("rb") as stream:
            x, y, width, height = map(int, stream.readline().split())
            values = np.fromfile(stream, dtype="<f4").reshape(height, width, 3)
        candidate = np.floor(np.clip(values, 0, 1) * 65535).astype(np.uint16)
        del values
    finally:
        output.unlink(missing_ok=True)
    current = decode(path)
    assert current.shape == candidate.shape and current.dtype == np.uint16
    assert np.any(current % 257)
    with rawpy.imread(str(path)) as raw:
        sizes = raw.sizes
        old = raw.postprocess(use_camera_wb=True, no_auto_bright=True, output_bps=16)
    assert old.shape != current.shape
    row = y - sizes.top_margin
    col = x - sizes.left_margin
    old_overlap = old[row : row + height, col : col + width]
    old_metrics = measures(candidate, old_overlap)
    corrected = measures(candidate, current)
    assert (
        corrected["mean_absolute_normalized_rgb"]
        < old_metrics["mean_absolute_normalized_rgb"]
    )
    assert sha(path) == frozen["sha256"]
    result["frames"][path.name] = {
        "source_sha256": frozen["sha256"],
        "geometry_metadata": geometry(path),
        "corrected_rgb_sha256": rgb_sha(current),
        "reference_historical_shape": list(old.shape),
        "candidate_rawler_shape": list(candidate.shape),
        "historical_common_overlap": old_metrics,
        "corrected_full_frame": corrected,
    }
    del candidate, current, old, old_overlap

regular = decode(scratch / "sources/frame_000.CR3")
portrait_path = scratch / "sources/portrait.CR3"
portrait = decode(portrait_path)
np.testing.assert_array_equal(portrait, np.rot90(regular, -1, axes=(0, 1)))
result["orientation"] = {
    "edited_file_sha256": sha(portrait_path),
    "edited_metadata": geometry(portrait_path),
    "native_shape": list(regular.shape),
    "rotated_shape": list(portrait.shape),
    "exactly_once_pixel_match": True,
}

(scratch / "capture-comparison.json").write_text(json.dumps(result, indent=2) + "\n")
