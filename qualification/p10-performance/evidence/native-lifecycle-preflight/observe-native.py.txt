"""Check real native lifecycle state availability without timing qualification."""
import json
import os
import shutil
import subprocess
import time
from pathlib import Path

from framebuffer import XTestPointer, XwdScreen, observed_desktop
from linux_latency import _prepare_pointer, _state, _wait_state
from pancake_conformance.fixtures import generate_fixture
from pancake_conformance.linux_ui import LinuxUISession
from pancake_conformance.macos_ui import MacOSUISession
from protocol import sha256

BASE=Path('artifacts/p10-native-lifecycle-preflight').resolve()
BINARY=BASE/'pancake-gui'
OUTPUT=BASE/'private-reviewed'
STATUS={'control':'stack.status','field':'title'}

def tap(ui,pointer,control):
    with _prepare_pointer(pointer,ui,{'kind':'click','at':{'control':control}}) as dispatch:
        dispatch()

def main():
    fixture_root=BASE/'fixture'
    fixture=generate_fixture('lifecycle',fixture_root)
    inputs=[str(path) for path in fixture.paths]
    argv=[str(BINARY),*inputs]
    report={
        'scope':'untimed native lifecycle observer availability diagnostic',
        'revision':'0859deb124ed9c9dc4a8dd12a16d309b9af321fc','binary_sha256':sha256(BINARY),
        'cargo_lock_sha256':'b2fa570aaa2dbc0265d938f9cb4567c78fe6ca2a0b8c6acba7c98fc381ec7cf4',
        'fixture_shape':fixture.shape,'source_count':len(inputs),'first_source_sha256':sha256(fixture.paths[0]),
        'arguments':argv,'observed_states':[], 'renderer':'winit-software',
    }
    try:
        env=dict(os.environ,SLINT_BACKEND='winit-software')
        with (
            observed_desktop(OUTPUT,env) as (private,path),
            XwdScreen(path) as screen,
            XTestPointer(private,screen) as pointer,
            (OUTPUT/'gui.stdout').open('w') as stdout,
            (OUTPUT/'gui.stderr').open('w') as stderr,
        ):
            process=subprocess.Popen(argv,env=private,stdout=stdout,stderr=stderr,start_new_session=True)
            try:
                with LinuxUISession(process,OUTPUT,private) as ui:
                    ui.click('method.laplacian')
                    ui.select('alignment','Breathing',timeout=15)
                    tap(ui,pointer,'stack.start')
                    _,station=_wait_state(ui,STATUS,{'running'},15)
                    report['observed_states'].append(station)
                    tap(ui,pointer,'stack.pause')
                    _,station=_wait_state(ui,STATUS,{'paused'},15)
                    report['observed_states'].append(station)
                    tap(ui,pointer,'stack.resume')
                    _,station=_wait_state(ui,STATUS,{'running'},15)
                    report['observed_states'].append(station)
                    tap(ui,pointer,'stack.stop')
                    deadline=time.monotonic()+30
                    phase=''
                    while time.monotonic()<deadline:
                        phase,station=_state(ui,STATUS)
                        if not report['observed_states'] or station!=report['observed_states'][-1]:
                            report['observed_states'].append(station)
                        if phase=='idle':
                            break
                    if phase!='idle':
                        raise TimeoutError('worker did not settle to native Idle')
                    report['native_cancelling_was_observed']=any(state.startswith('Cancelling') for state in report['observed_states'])
                    report['native_restart_enabled']=ui.node('stack.start').get('enabled')
                    report['status']='observed'
            finally:
                MacOSUISession._reap(process)
    except (AssertionError,OSError,RuntimeError,ValueError) as error:
        report['status']='incomplete'
        report['error']=str(error)
    finally:
        shutil.rmtree(fixture_root)
        report['temporary_fixture_removed']=not fixture_root.exists()
        (BASE/'receipt-checked.json').write_text(json.dumps(report,indent=2)+'\n')
    print(json.dumps({k:report[k] for k in ('status','error','observed_states','native_cancelling_was_observed','native_restart_enabled') if k in report}))

if __name__=='__main__':
    main()
