"""Remove only manifest-verified temporary P2 real corpus data after its last use."""

import argparse
import json
from collections import Counter
from pathlib import Path

from p10 import ADAPTERS, ROOT, TARGETS_PATH
from protocol import Campaign, corpus, sha256, verify_frames


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("corpus_root", type=Path)
    parser.add_argument("receipt", type=Path)
    args = parser.parse_args()
    root = args.corpus_root.resolve(strict=True)
    receipt = args.receipt.resolve()
    if args.corpus_root.is_symlink() or receipt.is_relative_to(root):
        raise ValueError(
            "require an actual owned corpus and an external cleanup receipt"
        )
    evidence = ROOT / "pancake/docs/p2-corpus-evidence"
    verification = evidence / "real-validation.json"
    acquisition = evidence / "real-acquisition.json"
    marker = root / "ownership.json"
    owner = json.loads(marker.read_text())
    if (
        owner.get("owner") != "codex/pancake-p2-corpus"
        or owner.get("verified_real_receipt_sha256") != sha256(verification)
        or set(owner.get("managed_directories", ()))
        != {"archives", "datasets", "source-metadata"}
    ):
        raise ValueError("cleanup marker does not match the retained corpus receipt")

    items = sorted(root.rglob("*"))
    if any(
        item.is_symlink() or not (item.is_dir() or item.is_file()) for item in items
    ):
        raise ValueError("unexpected special file in owned temporary corpus")
    verified = []
    campaign = Campaign.load(TARGETS_PATH)
    for workload, count in (("R50", 50), ("D150", 150), ("E9", 9)):
        rows, _ = corpus(ROOT, campaign, workload)
        paths = verify_frames(root, rows, count)
        verified.extend(
            {
                "role": workload,
                "path": str(path.relative_to(root)),
                "sha256": row["sha256"],
                "bytes": path.stat().st_size,
            }
            for path, row in zip(paths, rows, strict=True)
        )
    raw_manifest = json.loads(
        (ADAPTERS / "pancake/performance-raw-corpus.json").read_text()
    )
    metadata = json.loads(acquisition.read_text())["sources"]
    expected = {
        "archives/26E_dorsal_60x.zip": metadata["D150"]["archive_sha256"],
        "archives/MEFDatabase.rar": metadata["E9"]["archive_sha256"],
        "source-metadata/D02-LICENSE": raw_manifest["source"]["license_sha256"],
        "source-metadata/D02-README.md": raw_manifest["source"]["readme_sha256"],
        "source-metadata/D03.json": metadata["D150"]["metadata_sha256"],
        "source-metadata/D28.html": metadata["E9"]["metadata_sha256"],
    }
    for relative, digest in expected.items():
        path = root / relative
        if sha256(path) != digest:
            raise ValueError(f"temporary corpus metadata changed: {relative}")
        verified.append(
            {
                "role": relative.split("/", 1)[0],
                "path": relative,
                "sha256": digest,
                "bytes": path.stat().st_size,
            }
        )
    expected_files = {row["path"] for row in verified} | {marker.name}
    actual_files = {str(path.relative_to(root)) for path in items if path.is_file()}
    if expected_files != actual_files:
        raise ValueError("owned corpus contains unexpected files")

    result = {
        "root": str(root),
        "ownership_sha256": sha256(marker),
        "acquisition_receipt_sha256": sha256(acquisition),
        "verification_receipt_sha256": sha256(verification),
        "preflight_file_counts": dict(Counter(row["role"] for row in verified)),
        "preflight_verified_files": verified,
        "removed": [],
    }
    with receipt.open("x") as stream:
        try:
            for row in verified:
                (root / row["path"]).unlink()
                result["removed"].append(row["path"])
            for directory in sorted(
                (path for path in items if path.is_dir()),
                key=lambda path: len(path.parts),
                reverse=True,
            ):
                directory.rmdir()
            marker.unlink()
            root.rmdir()
        finally:
            result["owned_root_removed"] = not root.exists()
            result["removed_input_bytes"] = sum(
                row["bytes"] for row in verified[: len(result["removed"])]
            )
            json.dump(result, stream, indent=2)
            stream.write("\n")


if __name__ == "__main__":
    main()
