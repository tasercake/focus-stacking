#!/usr/bin/env bash
set -eu
root=$(git rev-parse --show-toplevel)
scope="$root/artifacts/e9-corrected-cuda"
qual=/home/krishna/Documents/focus-stacking/pancake-conformance/.venv/bin/python
producer=/home/krishna/Documents/focus-stacking/artifacts/pancake-linux-native/c1e2cac/pancake-cli
if test "$(git rev-parse HEAD)" != fb9c399a96658331cb76b843a1e565b503e94fdd; then
  printf '%s\n' 'Adjusted benchmark source must be audited again.' >&2
  exit 1
fi
test ! -e "$scope/paired"
test ! -e "$scope/window-start.txt"
export PYTHONPATH="$root/pancake-conformance/src:$root/qualification/p10-performance"
export PYTHONDONTWRITEBYTECODE=1 OPENBLAS_NUM_THREADS=1
{
  date --iso-8601=seconds
  uname -srmo
  cat /proc/loadavg
  ps -eo pid,stat,psr,%cpu,comm --sort=-%cpu | head -25
} > "$scope/window-start.txt"
status=0
env -u LD_LIBRARY_PATH taskset -c 0-31 "$qual" "$scope/bind_c1_native.py" \
  --run --producer-binary "$producer" --input-root "$scope/real/input" \
  --receipt "$scope/candidate-attachment.json" --output "$scope/paired" \
  > "$scope/run.stdout" 2> "$scope/run.stderr" || status=$?
{
  date --iso-8601=seconds
  printf 'exit=%s\n' "$status"
  cat /proc/loadavg
  ps -eo pid,stat,psr,%cpu,comm --sort=-%cpu | head -25
} > "$scope/window-end.txt"
printf 'campaign exit=%s\n' "$status"
if test "$status" -ne 0; then tail -30 "$scope/run.stderr"; fi
exit "$status"
