#!/usr/bin/env bash
set -eu
root=$(git rev-parse --show-toplevel)
cd "$root"
scope="$root/artifacts/e9-cpu-11cfcf8-3ms"
source_root=/tmp/pancake-worktrees.t3paqgWM/e9-cuda-integrated-replay/artifacts/e9-cuda-integrated-replay/input-stage/input
qual=/home/krishna/Documents/focus-stacking/pancake-conformance/.venv/bin/python
test "$(git rev-parse HEAD)" = 11cfcf8aac5921dfa843bfc5777eb4b2643a6ca6
test -z "$(git status --porcelain)"
test -f "$scope/no-device-preflight.json"
test -f "$scope/strict-ledger-prereq.json"
test -f "$scope/quiet-grant.json"
test ! -e "$scope/paired"
test ! -e "$scope/window-start.txt"
export PYTHONPATH="$scope:$root/pancake-conformance/src:$root/qualification/p10-performance"
export PYTHONDONTWRITEBYTECODE=1 OPENBLAS_NUM_THREADS=1
"$qual" -B "$scope/verify-quiet.py"
{
 date --iso-8601=seconds
 uname -srmo
 cat /proc/loadavg
 ps -eo pid,stat,psr,%cpu,comm --sort=-%cpu | head -25
} > "$scope/window-start.txt"
status=0
env -u LD_LIBRARY_PATH -u LD_PRELOAD taskset -c 0-31 "$qual" \
 "$scope/p10.py" run --profile E9.exposure_fusion.cpu \
 --runtime cpu --rss-sample-interval-ms 3 --threads 32 --timeout 120 --input-root "$source_root" \
 --build-receipt "$scope/build-receipt.json" --keep-review-images --output "$scope/paired" \
 > "$scope/run.stdout" 2> "$scope/run.stderr" || status=$?
{
 date --iso-8601=seconds
 printf 'exit=%s\n' "$status"
 cat /proc/loadavg
} > "$scope/window-end.txt"
printf 'campaign exit=%s\n' "$status"
if test "$status" -ne 0; then tail -25 "$scope/run.stderr"; fi
exit "$status"
