def require_metadata():
    for path, digest in TOOL_PINS.items():
        if sha256(path) != digest:
            raise ValueError(f"known collector or receiver changed: {path}")
    return runpy.run_path(str(EVIDENCE / "t50-weighted-triplet-plan/weighted-cpu-triplet.py.txt"))


def candidate_target(root, package_dir):
    # Activate only the retained extracted Debian executable, never the standalone ELF.
    root, package_dir = root.resolve(strict=True), package_dir.resolve(strict=True)
    descriptor = p10.CANDIDATE.relative_to(p10.ROOT)
    spec = TargetSpec.load(root / descriptor)
    p10._require_clean(root, "pancake", str(descriptor))
    snapshot = p10._snapshot(spec, spec.root / "Cargo.lock")
    if snapshot != {"commit": PACKAGE_SOURCE, "lock_sha256": LOCK, "descriptor_sha256": DESCRIPTOR}:
        raise ValueError("exact clean Debian producer checkout is required")
    if p10._git(root, "rev-parse", "HEAD^{tree}") != PACKAGE_TREE:
        raise ValueError("Debian source tree changed")
    verified = pinned_json(package_dir / "package-verification.json", PROOF)
    package = package_dir / "pancake_0.1.0_amd64.deb"
    candidate = package_dir / "verified-elf/pancake-cli"
    archive = {"file": package.name, "sha256": PKG, "bytes": 69_507_550}
    binary = {"sha256": ELF, "bytes": 122_323_792}
    if (verified["source_commit"] != PACKAGE_SOURCE or verified["cargo_lock_sha256"] != LOCK
            or verified["archive"] != archive or verified["binaries"]["pancake-cli"] != binary
            or package.stat().st_size != archive["bytes"] or sha256(package) != PKG
            or candidate.stat().st_size != binary["bytes"] or sha256(candidate) != ELF
            or not os.access(candidate, os.X_OK)):
        raise ValueError("verified Debian package or exact extracted CLI changed")
    return spec, snapshot | {
        "command": [str(candidate), *spec.command("cli")[1:]], "package": str(package),
        "package_sha256": PKG, "package_proof_sha256": PROOF,
        "binary_sha256": ELF, "tree": PACKAGE_TREE,
    }


def run_job(entry, root, deadline, target, profile, frames, receiver, rss_check):
    kind, number, role = entry
    spec, identity = target
    directory = root / f"{kind}-{number:02}" / role
    directory.mkdir(parents=True)
    cache = root / "cache" / role / ("warm" if kind in ("warm", "warmup") else "cold-01")
    output = directory / "complete.tif"
    env = p10._environment(spec, cache, directory, 15, None, "cpu" if role == "candidate" else None, True)
    env["LC_ALL"] = "C"
    argv = [*identity["command"], *spec.stack_arguments(frames, output, profile.options)]
    launch = {"kind": kind, "number": number, "target": role, "argv": argv,
              "cwd": str(spec.root), "environment": env, "sample_interval_ns": 3_000_000,
              "rss_sampler_cpu": 15, "cache_before": p10._cache_inventory(cache),
              "cpu_temperature_before_millidegrees": p10._cpu_thermal_state()}
    if kind in ("warmup", "cold") and launch["cache_before"]["files"]:
        raise ValueError("first warmup and cold job must use empty private caches")
    p10._write(directory / "launch.json", receiver["sanitized"](launch))
    timeout = min(120, deadline - time.monotonic())
    if timeout <= 0:
        raise ValueError("bounded campaign deadline exhausted")
    sample = launch | {"valid": False}
    try:
        sample.update(resources.measure(argv, cwd=spec.root, env=env, directory=directory,
                                        timeout=timeout, gpu=None, rss_sampler_cpu=15,
                                        rss_sample_interval_ms=3))
        if sample["returncode"] or sample["timed_out"] or sample["resource_errors"]:
            raise ValueError("job failed, timed out, or leaked resources")
        sample["cpu_temperature_after_millidegrees"] = p10._cpu_thermal_state()
        sample["cache_after"] = p10._cache_inventory(cache)
        image = verify_output(output, (4000, 6000, 3))
        completion = p10._assert_completion(directory / "events.jsonl", "cpu",
                                            "cpu" if role == "candidate" else None, None)
        events = [json.loads(line) for line in (directory / "events.jsonl").read_text().splitlines()]
        if receiver["check_completion"](role, events, frames, image) != completion:
            raise ValueError("independent complete-job map or crop witness disagreed")
        if (image["sha256"], image["pixel_sha256"]) != EXPECTED_OUTPUTS[role]:
            raise ValueError("real new RGB16 bytes or pixels require separate quality review")
        sample.update(valid=True, output=image, completion=completion)
        with (directory / "resources.jsonl").open() as raw:
            sample["rss_check"] = rss_check(role, {
                "sample": sample, "launch": launch, "observations": [json.loads(line) for line in raw],
            })
    except (KeyError, OSError, RuntimeError, ValueError) as error:
        sample.update(valid=False, invalid_reason=str(error))
        raise
    finally:
        p10._write(directory / "sample.json", receiver["sanitized"](sample))
        p10._compress_trace(directory)
    return receiver["sanitized"](sample), output


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--execute-limited", action="store_true")
    parser.add_argument("--lease", type=Path)
    parser.add_argument("--package-dir", type=Path)
    parser.add_argument("--producer-source", type=Path)
    parser.add_argument("--output", type=Path)
    args = parser.parse_args()
    receiver = require_metadata()
    policy = runpy.run_path(str(HERE / "limited-gates.py.txt"))
    envelope = runpy.run_path(str(EVIDENCE / "t50-memory-calibration-plan/calibration-envelope.py.txt"))
    campaign, profile, stage, frames, provenance = receiver["settings"]()
    schedule = policy["SCHEDULE"]
    description = {
        "frozen_full_credit": False, "profile": profile.key, "settings": profile.options,
        "schedule": [list(row) for row in schedule], "limits": policy["LIMITS"], "sample_interval_ns": 3_000_000,
        "materialization_sha256": stage["materialization_sha256"], "frames": len(frames),
        "cli_sha256": ELF, "package_sha256": PKG, "collector_tools": {str(path): sha for path, sha in TOOL_PINS.items()},
    }
    if not args.execute_limited:
        print(json.dumps(description | {"scope": "static preregistration only"}, indent=2))
        return 0
    if not all((args.lease, args.output, args.producer_source, args.package_dir)):
        parser.error("execution requires an explicit new quiet/custody lease, producers and fresh output")
    lease = json.loads(args.lease.read_bytes())
    if any(lease.get(key) is not True for key in ("exclusive_host_granted", "stage6_read_granted",
                                                  "other_workloads_terminal")) or any(
        lease.get(key) != description[key] for key in ("profile", "sample_interval_ns", "materialization_sha256", "schedule")
    ):
        raise ValueError("lease does not authorize this exact bounded schedule")
    if os.sched_getaffinity(0) != set(range(15)):
        raise ValueError("expected only worker CPUs 0–14")
    root = args.output.resolve()
    protected = (Path(stage["input_root"]).parent, args.producer_source, args.package_dir,
                 receiver["reference_spec"]()[0].root)
    if any(root.is_relative_to(path.resolve()) for path in protected):
        raise ValueError("output must stay outside source, package, reference and shared Stage6")
    if shutil.disk_usage(root.parent).free < 15 * 1024**3:
        raise ValueError("need at least 15 GiB free for proof retention")
    root.mkdir(mode=0o700)
    report = description | {"scope": "bounded limited measurement only", "jobs": [],
                             "started_unix_ns": time.time_ns(), "lease_sha256": sha256(args.lease),
                             "operator_sha256": sha256(Path(__file__)),
                             "limited_reducer_sha256": sha256(HERE / "limited-gates.py.txt")}
    receipt = root / "report.json"
    try:
        with p10._campaign_lock():
            reference = receiver["bind_reference"](campaign)
            candidate = candidate_target(args.producer_source, args.package_dir)
            targets = {"reference": reference, "candidate": candidate}
            if reference[1]["command"][1:] != ["-m", "src.cli"]:
                raise ValueError("require pinned normal ChimpStackr executable entrypoint")
            report.update(targets={role: row[1] for role, row in targets.items()}, hardware=p10._hardware(None))
            original = receiver["authenticate_stage"](stage, provenance, frames)
            exemplar = {}
            deadline = time.monotonic() + 600
            try:
                for entry in schedule:
                    _, _, role = entry
                    sample, output = run_job(entry, root, deadline, targets[role], profile, frames,
                                             receiver, envelope["arm_rss"])
                    report["jobs"].append(sample)
                    exemplar.setdefault(role, output)
                    p10._write(receipt, report)
                report["limited_numeric"] = policy["reduce_gates"](report["jobs"])
                oracle = runpy.run_path(str(EVIDENCE / "registration-recheck/scale-recheck.py.txt"))
                truth = {role: oracle["generated_oracle"](path, Path(stage["input_root"]), campaign)
                         for role, path in exemplar.items()}
                if any([row["depth"] for row in value["regions"]] != list(range(50)) for value in truth.values()):
                    raise ValueError("generated sharp-truth coverage incomplete")
                report["quality"] = {
                    "truth": truth, "all_exports_reproduce_prior_reviewed_bytes_and_pixels": True,
                    "paired_pixels": pixel_metrics(exemplar["reference"], exemplar["candidate"], (4000, 6000, 3)),
                }
            finally:
                report["all_52_unchanged"] = receiver["authenticate_stage"](stage, provenance, frames) == original
                report["producers_unchanged"] = (
                    receiver["bind_reference"](campaign)[1] == reference[1]
                    and candidate_target(args.producer_source, args.package_dir)[1] == candidate[1]
                )
            if not report["all_52_unchanged"] or not report["producers_unchanged"]:
                raise ValueError("source or producer changed")
        report["status"] = report["limited_numeric"]["status"]
    except (KeyError, OSError, RuntimeError, ValueError) as error:
        report.update(status="invalid", error=str(error))
        raise
    finally:
        report["finished_unix_ns"] = time.time_ns()
        p10._write(receipt, report)
    return report["status"] != "pass"


if __name__ == "__main__":
    raise SystemExit(main())
