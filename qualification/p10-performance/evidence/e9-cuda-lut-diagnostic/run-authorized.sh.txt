#!/usr/bin/env bash
set -eu
root=$(git rev-parse --show-toplevel)
cd "$root"
scope="$root/artifacts/e9-cuda-lut-diagnostic"
proof="$scope/candidate-evidence.json"
qual=/home/krishna/Documents/focus-stacking/pancake-conformance/.venv/bin/python
script="$root/qualification/p10-performance/evidence/e9-cuda-lut-diagnostic/measure.py.txt"
proof_sha=943593942c238383516046d086ad1404b8232590f8934d0ecbeb712efc844ee6
printf '%s  %s\n' "$proof_sha" "$proof" | sha256sum --check --status
test ! -e "$scope/session"
test ! -e "$scope/window-start.txt"
export PYTHONPATH="$root/pancake-conformance/src:$root/qualification/p10-performance"
export PYTHONDONTWRITEBYTECODE=1 OPENBLAS_NUM_THREADS=1
{
  date --iso-8601=seconds
  uname -srmo
  cat /proc/loadavg
  ps -eo pid,stat,psr,%cpu,comm --sort=-%cpu | head -20
} > "$scope/window-start.txt"
status=0
env -u LD_LIBRARY_PATH -u LD_PRELOAD timeout --signal=TERM --kill-after=5s 120s \
  taskset -c 0-31 "$qual" "$script" run --input-root "$scope/input-stage/input" \
  --stock "$scope/binaries/stock" --lut "$scope/binaries/lut" --candidate-evidence "$proof" \
  --candidate-evidence-sha256 "$proof_sha" --output "$scope/session" \
  > "$scope/run.stdout" 2> "$scope/run.stderr" || status=$?
{
  date --iso-8601=seconds
  printf 'exit=%s\n' "$status"
  cat /proc/loadavg
} > "$scope/window-end.txt"
printf 'diagnostic exit=%s\n' "$status"
if test "$status" -ne 0; then tail -25 "$scope/run.stderr"; fi
exit "$status"
